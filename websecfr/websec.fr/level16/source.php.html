function gcd($n, $m) {
    if (!ctype_digit ($n) or !ctype_digit ($m))
        return 1;
    while(true) {
        if($n == $m) {
            return $m;
        } elseif($n > $m) {
            $n -= $m;
        } else {
            $m -= $n;
        }
    }
}
